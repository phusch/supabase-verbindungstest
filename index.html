<script type="module">

import { createClient } from

  "https://esm.sh/@supabase/supabase-js@2";


const projectUrl =

  "https://qijordwpbwwbizfcslrv.supabase.co";


const expectedUID =

  "9be7d1bc-4d69-4cb8-bb42-cb2d3c705f75";


let supabase = null;

const $ = id => document.getElementById(id);


function show(text) {

  $("result").textContent = text;

}


function state(text) {

  $("status").textContent = text;

}


function lock(ready) {

  $("test").disabled = !ready;

  $("logout").disabled = !ready;

}


$("login").onclick = async () => {

  $("login").disabled = true;

  lock(false);

  state("🟡 Anmeldung läuft...");


  try {

    const key = $("key").value.trim();

    if (!key.startsWith("sb_publishable_")) {

      throw new Error(

        "Bitte Publishable Key eintragen."

      );

    }


    supabase = createClient(projectUrl, key, {

      auth: {

        persistSession: false,

        autoRefreshToken: false,

        detectSessionInUrl: false

      }

    });


    const {data,error} =

      await supabase.auth.signInWithPassword({

        email: $("email").value.trim(),

        password: $("password").value

      });


    $("password").value = "";


    if (error) throw error;


    if (data.user?.id !== expectedUID) {

      await supabase.auth.signOut();

      throw new Error("Benutzer-ID stimmt nicht.");

    }


    state("🟢 Supabase-Anmeldung erfolgreich");

    show("Angemeldet. Jetzt GitHub testen.");

    lock(true);

  } catch(e) {

    $("password").value = "";

    state("🔴 Anmeldung fehlgeschlagen");

    show(e.message);

  } finally {

    $("login").disabled = false;

  }

};


$("test").onclick = async () => {

  $("test").disabled = true;

  state("🟡 GitHub wird geprüft...");


  try {

    const {data,error} =

      await supabase.functions.invoke(

        "github-app-storage",

        {body:{}}

      );


    if (error) {

      let details = error.message;

      if (error.context instanceof Response) {

        const response = error.context;

        details += " (HTTP " + response.status + ")";

        details += "\n" + await response.text();

      }

      throw new Error(details);

    }


    show(JSON.stringify(data,null,2));


    if (data?.ok &&

        data?.githubConnected === true &&

        data?.private === true &&

        data?.repository ===

          "phusch/huschel-app-data") {

      state("🟢 GitHub-Verbindung erfolgreich");

    } else {

      state("🔴 GitHub-Prüfung fehlgeschlagen");

    }

  } catch(e) {

    state("🔴 Verbindungstest fehlgeschlagen");

    show(e.message);

  } finally {

    $("test").disabled = false;

  }

};


$("logout").onclick = async () => {

  if (supabase) await supabase.auth.signOut();

  lock(false);

  state("🟡 Abgemeldet");

  show("Sitzung beendet.");

};
